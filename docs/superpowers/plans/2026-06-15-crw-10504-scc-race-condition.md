# CRW-10504: SCC Permission Race Condition Fix — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Prevent first-time users from hitting "user is not permitted use the container-build SCC" by adding a pre-flight SCC permission check to the workspace creation flow.

**Architecture:** New backend endpoint performs a `SelfSubjectAccessReview` with the user's token to check if they can `use` the target SCC. A new frontend progress step polls this endpoint before the SCC attribute patch. For returning users the first poll returns `permitted: true` — zero delay. Existing `getCurrentScc()` helper and `selectCurrentScc` selector are reused for SCC name resolution.

**Tech Stack:** @kubernetes/client-node `AuthorizationV1Api`, Fastify, React class component (extends `ProgressStep`), Redux `connect()`, Jest

// Generated by Claude Opus 4.6

import * as k8s from '@kubernetes/client-node';

import { createError } from '@/devworkspaceClient/services/helpers/createError';
import { ISccPermissionApi } from '@/devworkspaceClient/types';

const ERROR_LABEL = 'SCC_PERMISSION_API_ERROR';

export class SccPermissionApiService implements ISccPermissionApi {
  private readonly authorizationApi: k8s.AuthorizationV1Api;

  constructor(kc: k8s.KubeConfig) {
    this.authorizationApi = kc.makeApiClient(k8s.AuthorizationV1Api);
  }

  async checkSccPermission(namespace: string, sccName: string): Promise<boolean> {
    try {
      const review: k8s.V1SelfSubjectAccessReview = {
        apiVersion: 'authorization.k8s.io/v1',
        kind: 'SelfSubjectAccessReview',
        spec: {
          resourceAttributes: {
            namespace,
            verb: 'use',
            group: 'security.openshift.io',
            resource: 'securitycontextconstraints',
            name: sccName,
          },
        },
      };

      const response = await this.authorizationApi.createSelfSubjectAccessReview(review);
      return response.body.status?.allowed === true;
    } catch (error) {
      throw createError(error, ERROR_LABEL, 'Unable to check SCC permission');
    }
  }
}

import { FastifyInstance, FastifyRequest } from 'fastify';

import { baseApiPath } from '@/constants/config';
import { getDevWorkspaceClient } from '@/routes/api/helpers/getDevWorkspaceClient';
import { getToken } from '@/routes/api/helpers/getToken';
import { getSchema } from '@/services/helpers';

const tags = ['SCC Permission'];

export function registerSccPermissionRoute(instance: FastifyInstance) {
  instance.register(async server => {
    server.get(
      `${baseApiPath}/namespace/:namespace/scc-permission/:scc`,
      getSchema({
        tags,
        params: {
          type: 'object',
          properties: {
            namespace: { type: 'string' },
            scc: { type: 'string' },
          },
          required: ['namespace', 'scc'],
        },
      }),
      async function (request: FastifyRequest) {
        const { namespace, scc } = request.params as { namespace: string; scc: string };
        const token = getToken(request);

        const { sccPermissionApi } = getDevWorkspaceClient(token);
        const permitted = await sccPermissionApi.checkSccPermission(namespace, scc);

        return { permitted };
      },
    );
  });
}

// Generated by Claude Opus 4.6

import { helpers } from '@eclipse-che/common';
import { AlertVariant } from '@patternfly/react-core';
import isEqual from 'lodash/isEqual';
import React from 'react';
import { connect, ConnectedProps } from 'react-redux';

import {
  ProgressStep,
  ProgressStepProps,
  ProgressStepState,
} from '@/components/WorkspaceProgress/ProgressStep';
import { ProgressStepTitle } from '@/components/WorkspaceProgress/StepTitle';
import { checkSccPermission } from '@/services/backend-client/sccPermissionApi';
import { AlertItem } from '@/services/helpers/types';
import { RootState } from '@/store';
import { selectDefaultNamespace } from '@/store/InfrastructureNamespaces/selectors';
import { selectCurrentScc } from '@/store/ServerConfig/selectors';

const SCC_POLL_INTERVAL_MS = 2000;
const SCC_POLL_TIMEOUT_MS = 60000;

export type Props = MappedProps & ProgressStepProps;
export type State = ProgressStepState;

class CreatingStepCheckSccPermission extends ProgressStep<Props, State> {
  protected readonly name = 'Waiting for namespace provisioning to complete';

  constructor(props: Props) {
    super(props);
    this.state = {
      name: this.name,
    };
  }

  public componentDidMount() {
    this.init();
  }

  public componentDidUpdate() {
    this.init();
  }

  public shouldComponentUpdate(nextProps: Props, nextState: State): boolean {
    if (this.props.distance !== nextProps.distance) {
      return true;
    }
    if (!isEqual(this.state.lastError, nextState.lastError)) {
      return true;
    }
    return false;
  }

  private init() {
    if (this.props.distance !== 0) {
      return;
    }
    if (this.state.lastError) {
      return;
    }
    this.prepareAndRun();
  }

  protected handleRestart(): void {
    this.clearStepError();
    this.props.onRestart();
  }

  protected async runStep(): Promise<boolean> {
    const { currentScc, defaultNamespace } = this.props;

    if (!currentScc) {
      return true;
    }

    const namespace = defaultNamespace.name;
    const startTime = Date.now();

    while (Date.now() - startTime < SCC_POLL_TIMEOUT_MS) {
      try {
        const { permitted } = await checkSccPermission(namespace, currentScc);
        if (permitted) {
          return true;
        }
      } catch (e) {
        console.warn('SCC permission check failed, proceeding anyway:', e);
        return true;
      }

      await new Promise(resolve => setTimeout(resolve, SCC_POLL_INTERVAL_MS));
    }

    throw new Error(
      'Namespace provisioning did not complete in time. Please try again in a moment.',
    );
  }

  protected buildAlertItem(error: Error): AlertItem {
    return {
      key: this.name,
      title: 'Warning: namespace provisioning timeout',
      variant: AlertVariant.warning,
      children: helpers.errors.getMessage(error),
      actionCallbacks: [
        {
          title: 'Restart',
          callback: () => this.handleRestart(),
        },
      ],
    };
  }

  render(): React.ReactElement {
    const { distance, hasChildren } = this.props;
    const { name, lastError } = this.state;

    const isError = lastError !== undefined;

    return (
      <React.Fragment>
        <ProgressStepTitle
          distance={distance}
          hasChildren={hasChildren}
          isError={isError}
          isWarning={false}
          parentStepName="Creating a workspace"
        >
          {name}
        </ProgressStepTitle>
      </React.Fragment>
    );
  }
}

const mapStateToProps = (state: RootState) => ({
  currentScc: selectCurrentScc(state),
  defaultNamespace: selectDefaultNamespace(state),
});

const connector = connect(mapStateToProps, null, null, {
  forwardRef: true,
});
type MappedProps = ConnectedProps<typeof connector>;
export default connector(CreatingStepCheckSccPermission);
